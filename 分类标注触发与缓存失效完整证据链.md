def matching_resources_scope
  scope = registry.resource_scope
  
  # 1. Prepare the query with joins required by conditions
  conditions.each do |condition|
    scope = condition.prepare(scope)
  end

  # 2. Apply the conditions to the query
  conditions.each do |condition|
    scope = condition.apply(scope)
  end

  scope
end

### 1.6 分类动作执行器

**文件位置**: `app/models/rule/action_executor/set_transaction_category.rb:10-26`

```ruby
def execute(transaction_scope, value: nil, ignore_attribute_locks: false)
  category = family.categories.find_by_id(value)

  scope = transaction_scope

  # 🔑 关键过滤：排除已被用户锁定的交易
  unless ignore_attribute_locks
    scope = scope.enrichable(:category_id)
  end

  scope.each do |txn|
    txn.enrich_attribute(
      :category_id,
      category.id,
      source: "rule"  # 标记来源为规则引擎
    )
  end
end
```

### 1.7 属性增强与状态变更

**文件位置**: `app/models/concerns/enrichable.rb:26-51`

```ruby
def enrich_attribute(attr, value, source:, metadata: {})
  enrich_attributes({ attr => value }, source: source, metadata: metadata)
end

def enrich_attributes(attrs, source:, metadata: {})
  # 🔑 防御性检查：仅处理未锁定且值有变化的属性
  enrichable_attrs = Array(attrs).reject do |attr_key, attr_value|
    locked?(attr_key) || ignored_enrichable_attributes.include?(attr_key) || self[attr_key.to_s] == attr_value
  end

  ActiveRecord::Base.transaction do
    enrichable_attrs.each do |attr, value|
      self.send("#{attr}=", value)

      unless self.new_record?
        # 记录变更历史
        log_enrichment(
          attribute_name: attr, 
          attribute_value: value, 
          source: source, 
          metadata: metadata
        )
      end
    end

    save  # 触发 ActiveRecord 回调，包括 touch
  end
end
```

### 2.3 属性锁定机制

**文件位置**: `app/models/concerns/enrichable.rb:69-73`

```ruby
def lock_saved_attributes!
  # 锁定所有发生变化的属性
  saved_changes.keys.reject do |attr| 
    ignored_enrichable_attributes.include?(attr) 
  end.each do |attr|
    lock_attr!(attr)
  end
end

def lock_attr!(attr)
  update!(locked_attributes: locked_attributes.merge(attr.to_s => Time.current))
end
```

**locked_attributes 结构**:
```json
{
  "category_id": "2026-05-16T10:30:00Z",
  "merchant_id": "2026-05-15T14:20:00Z"
}
```

### 2.4 时间戳传播链

#### 第一层：Entryable → Entry 自动 touch

**文件位置**: `app/models/entryable.rb:13`

```ruby
included do
  include Enrichable
  
  # 🔑 关键配置：entryable 变更自动触发 entry 时间戳更新
  has_one :entry, as: :entryable, touch: true
  
  # ... 其他 scope
end
```

**证据**：当 `Transaction.category_id` 变化时
1. `Transaction.updated_at` 更新
2. 自动触发 `Entry.updated_at` 更新（由 `touch: true` 保证）

#### 第二层：Entry → entries_cache_version 计算

**文件位置**: `app/models/family.rb:110-115`

```ruby
def entries_cache_version
  @entries_cache_version ||= begin
    ts = entries.maximum(:updated_at)
    ts.present? ? ts.to_i : 0  # 转换为 Unix 时间戳整数
  end
end
```

**计算逻辑**:
- 查询 `MAX(entries.updated_at)` 获取最新变更时间
- 转换为 Unix 时间戳整数作为缓存版本号
- 任何 entry 的更新都会导致版本号递增

### 2.5 缓存键设计与失效机制

**文件位置**: `app/models/income_statement.rb:103-120`

#### 缓存键结构

```ruby
# 分类统计缓存键
["income_statement", "category_stats", family_id, interval, entries_cache_version]

# 家庭统计缓存键
["income_statement", "family_stats", family_id, interval, entries_cache_version]

# 总计查询缓存键
["income_statement", "totals_query", family_id, sql_hash, entries_cache_version]
```

#### 失效原理

```
原始缓存键 (版本 V1):
["income_statement", "totals_query", 123, "abc123", 1715855400]

用户修改交易后，entries_cache_version 变为 V2:
1715855400 → 1715855467 (+67秒)

新缓存键 (版本 V2):
["income_statement", "totals_query", 123, "abc123", 1715855467]

🔑 结果：Rails.cache.fetch 检测到键不存在，自动执行 block 重新计算
```

def median_expense(interval: "month", category: nil)
  if category.present?
    category_stats(interval: interval).find do |stat|
      stat.classification == "expense" && stat.category_id == category.id
    end&.median || 0
  else
    family_stats(interval: interval).find do |stat|
      stat.classification == "expense"
    end&.median || 0
  end
end
